# 54. Spiral Matrix

Time Complexity: O(m*n)

Space Complexity: O(1)

Asked by Amazon:

keep four walls: `right`, `down`, `left` and `up`. Maintain an `enum for directions` .

Starting from `i=0`, `j=0`…. traverse through the particular direction till you hit the wall, bring the wall one step closer and then change directions, till you hit the wall in that direction. Repeat till you have collected `m*n` elements.

Note: At starting `up` will always start from 0, because we are always going to start from first row in left to right direction, so up wall will be pushed to 0th anyway after first left to right run.

```cpp
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int row = matrix.size();
        int col = matrix[0].size();

        vector<int> res;
        enum Directions { UP, RIGHT, DOWN, LEFT };
        enum Directions direction = RIGHT;

        int UP_WALL = 0; // not setting to -1, we will keep it incremented by beginning
        int RIGHT_WALL = col;
        int DOWN_WALL = row;
        int LEFT_WALL = -1;

        int i=0, j=0;
        int count = 0;

        while (count < row * col) {
            switch (direction) {
                case RIGHT:
                    while (j < RIGHT_WALL) {
                        res.push_back(matrix[i][j++]);
                        count++;
                    }
                    i++;
                    j--;
                    RIGHT_WALL--;
                    direction = DOWN;
                    break;

                case DOWN:
                    while (i < DOWN_WALL) {
                        res.push_back(matrix[i++][j]);
                        count++;
                    }
                    i--;
                    j--;
                    DOWN_WALL--;
                    direction = LEFT;
                    break;

                case LEFT:
                    while (j > LEFT_WALL) {
                        res.push_back(matrix[i][j--]);
                        count++;
                    }
                    i--;
                    j++;
                    LEFT_WALL++;
                    direction = UP;
                    break;

                case UP:
                    while (i > UP_WALL) {
                        res.push_back(matrix[i--][j]);
                        count++;
                    }
                    i++;
                    j++;
                    UP_WALL++;
                    direction = RIGHT;
            }
        }

        return res;
    }
};
```
