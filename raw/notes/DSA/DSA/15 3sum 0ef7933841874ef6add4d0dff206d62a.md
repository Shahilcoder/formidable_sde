# 15. 3sum

Nested loop and Two pointers approach:

Time complexity: O(N^2)

Space complexity: O(1)

Sort the array, run a loop over the array, and reduce the problem into two sum problem and apply two pointer approach on it.

But in order to avoid duplication, exactly and carefully follow the steps that are commented in code.

After repetitions ignored move ahead flow is VERY important.

LOOK AT NEETCODE’s EXPLANATION THAT’s MOST SIMPLE.

```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        int size = nums.size();
        vector<vector<int>> res;

        // sort the array in increasing order
        sort(nums.begin(), nums.end());

        int i = 0;
        while (i < size) {
            // first reduce it to two sum problem
            int a = nums[i];
            int t = -a; // b + c = -a

            // two pointers
            int s = i + 1, e = size - 1;

            while (s < e) {
                if ((nums[s] + nums[e]) == t) {
                    res.push_back({a, nums[s], nums[e]});
                    // ignore repeating values, we have found already triplets with these values
                    while ((s < e) && nums[s + 1] == nums[s]) s++;
                    while ((s < e) && nums[e - 1] == nums[e]) e--;

                    // after repetitions ignored, move ahead
                    s++;
                    e--;
                } else if ((nums[s] + nums[e]) > t) e--;
                else s++;
            }

            // ignore repeating values
            while((i+1 < size) && nums[i+1] == nums[i]) i++;

            // after repetitions ignored, move ahead
            i++;
        }

        return res;
    }
};
```

```cpp
class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        int size = nums.size();

        vector<vector<int>> res;

        sort(nums.begin(), nums.end());

        for (int i=0; i < size - 2; ++i) {
            int left = i + 1, right = size - 1;

            while (left < right) {
                int sum = nums[i] + nums[left] + nums[right];

                if (sum == 0) {
                    res.push_back({nums[i], nums[left], nums[right]});

                    while (left < right && nums[left] == nums[left + 1]) left++;
                    while (right > left && nums[right] == nums[right - 1]) right--;

                    left++;
                    right--;
                } else if (sum > 0) right--;
                else left++;
            }

            while (i < (size - 2) && nums[i] == nums[i+1]) i++;
        }

        return res;
    }
};
```
