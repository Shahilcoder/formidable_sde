# 55. Jump Game

My approach:

Time complexity: O(n) checked from leetcode

Space complexity: O(1)

start from second last index, traverse backwards till you find a 0. if there is a zero, then traverse backwards till you find element that can crossover this 0. If that’s not possible then return false.

If it is possible, continue with checking for if there is any other 0s, and do the same check if they can be crossed over.

If you reached beginning of array, means it's all good, you can return true.

```cpp
class Solution {
public:
    bool canJump(vector<int>& nums) {
        int size = nums.size();
        int index = size - 2;
        bool canJump = true;

        while (index >= 0) {
            if (nums[index] == 0) {
                canJump = false;
                for (int i = index - 1; i >=0; --i) {
                    if (nums[i] > (index - i)) {
                        canJump = true;
                        break;
                    }
                }

                if (!canJump) return canJump;
            }

            index--;
        }

        return canJump;
    }
};
```

class Solution {
public:
    bool canJump(vector<int>& nums) {
        int size = nums.size();

        int goal = size - 1;

        for (int i=size-2; i>=0; --i) {
            if (i + nums[i] >= goal) goal = i;
        }

        if (goal == 0) return true;
        return false;
    }
};
