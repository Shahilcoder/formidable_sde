# 1. Two Sum

First approach: Brute force,

Time complexity: O(n^2)

Space complexity: O(1)

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        if (nums.size() == 2) return {0, 1};

        for (int i=0; i < nums.size() - 1; ++i) {
            int to_search = target - nums[i];

            for (int j = i + 1; j < nums.size(); ++j) {
                if (nums[j] == to_search) return {i, j};
            }
        }

        return {-1, -1};
    }
};
```

Second approach: Using unordered_map (hash table inside, although official cppreference site tells its a bucket)

[https://en.cppreference.com/w/cpp/container/unordered_map/find](https://en.cppreference.com/w/cpp/container/unordered_map/find)

Time Complexity: O(n)

Space Complexity: O(n)

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        if (nums.size() == 2) return {0, 1};

        // {{key, value}, {key, value}}
        unordered_map<int, int> map;

        for (int i=0; i<nums.size(); ++i) map[nums[i]] = i;

        for (int i=0; i < nums.size(); ++i) {
            int to_search = target - nums[i];

            // check->first will be the key or the array element
            // check->second will be the value or the array index stored
            // O(1) in average case
            if (auto check = map.find(to_search); check != map.end() && check->second != i)
                return {i, check->second};
        }

        return {-1, -1};
    }
};
```

Third approach: Vector of pairs, with two pointer technique, (array also sorted)

[https://en.cppreference.com/w/cpp/algorithm/sort](https://en.cppreference.com/w/cpp/algorithm/sort)

Time complexity: O(nlogn) [don’t know about two pointer…]

Space complexity: O(n)

```cpp
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        if (nums.size() == 2) return {0, 1};

        vector<pair<int, int>> store;

        for (int i=0; i < nums.size(); ++i)
            store.push_back({nums[i], i});
        
        sort(store.begin(), store.end(), [](const pair<int, int> &a, const pair<int, int> &b) {
            // "<=" errors out as some memory error :(
            return a.first < b.first;
        });

        auto s = store.begin();
        auto e = store.end() - 1;

        while (s != e) {
            int sum = s->first + e->first;

            if (sum == target) return {s->second, e->second};
            else if (sum > target) e--;
            else s++;
        }

        return {-1, -1};
    }
};
```
